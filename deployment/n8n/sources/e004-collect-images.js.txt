function normalizeUserPath(value, options = {}) {
  const trimTrailingSlash = options.trimTrailingSlash !== false;
  let result = String(value ?? '').trim();
  if (result.length >= 2) {
    const first = result[0];
    const last = result[result.length - 1];
    if ((first === '"' && last === '"') || (first === "'" && last === "'")) result = result.slice(1, -1).trim();
  }
  result = result.replace(/\\+/g, '/').replace(/^([A-Za-z]:)\/+/, '$1/');
  if (trimTrailingSlash) while (result.length > 1 && result.endsWith('/') && !/^[A-Za-z]:\/$/.test(result)) result = result.slice(0, -1);
  return result;
}

function cleanName(value, fallback) {
  return String(value || fallback || 'product').trim().replace(/[<>:"/\\|?*\x00-\x1F]/g, '_').replace(/[. ]+$/g, '') || fallback || 'product';
}

function extensionOf(value) {
  const name = String(value || '').split('/').pop() || '';
  const index = name.lastIndexOf('.');
  return index >= 0 ? name.slice(index).toLowerCase() : '';
}

function normalizeRelativePath(value, label) {
  const normalized = normalizeUserPath(value, { trimTrailingSlash: false }).replace(/^\.\/+/, '');
  if (!normalized || normalized.startsWith('/') || /^[A-Za-z]:\//.test(normalized)) throw new Error(label + ' 必须是非空相对路径。');
  const segments = normalized.split('/');
  if (segments.some((segment) => !segment || segment === '.' || segment === '..')) throw new Error(label + ' 包含不安全的路径片段：' + normalized);
  return segments.join('/');
}

function resolveInside(root, relativePath, label) {
  const absoluteRoot = path.resolve(root);
  const absoluteTarget = path.resolve(absoluteRoot, ...relativePath.split('/'));
  const relative = path.relative(absoluteRoot, absoluteTarget);
  if (!relative || relative === '..' || relative.startsWith('..' + path.sep) || path.isAbsolute(relative)) throw new Error(label + ' 超出来源目录：' + relativePath);
  return absoluteTarget;
}

function relativeFromRoot(root, filePath, label) {
  const absoluteRoot = path.resolve(root);
  const absoluteFile = path.resolve(filePath);
  const relative = path.relative(absoluteRoot, absoluteFile);
  if (!relative || relative === '..' || relative.startsWith('..' + path.sep) || path.isAbsolute(relative)) throw new Error(label + ' 超出来源目录：' + filePath);
  return relative.split(path.sep).join('/');
}

function boundedInteger(value, label, minimum, maximum) {
  const number = Number(value);
  if (!Number.isInteger(number) || number < minimum || number > maximum) throw new Error(label + ' 必须是 ' + minimum + '–' + maximum + ' 的整数。');
  return number;
}

function orderManifestFiles(manifest) {
  if (!manifest || !Array.isArray(manifest.selectedFiles) || manifest.selectedFiles.length === 0) throw new Error('selection-manifest.json 的 selectedFiles 必须是非空数组。');
  const files = manifest.selectedFiles.map((entry, index) => {
    if (!entry || typeof entry !== 'object' || Array.isArray(entry)) throw new Error('selection-manifest.json 的 selectedFiles[' + index + '] 无效。');
    return { ...entry };
  });
  const hasSortOrder = files.map((entry) => Object.prototype.hasOwnProperty.call(entry, 'sortOrder'));
  if (hasSortOrder.some(Boolean) && !hasSortOrder.every(Boolean)) throw new Error('selection-manifest.json 的 sortOrder 不得部分缺失。');
  if (!hasSortOrder.every(Boolean)) return { files, orderSource: 'selection-manifest-array' };
  const seen = new Set();
  for (const [index, entry] of files.entries()) {
    if (!Number.isInteger(entry.sortOrder) || entry.sortOrder < 0) throw new Error('selection-manifest.json 的 selectedFiles[' + index + '].sortOrder 必须是非负整数。');
    if (seen.has(entry.sortOrder)) throw new Error('selection-manifest.json 的 sortOrder 重复：' + entry.sortOrder);
    seen.add(entry.sortOrder);
  }
  const ordered = [...files].sort((left, right) => left.sortOrder - right.sortOrder);
  ordered.forEach((entry, index) => {
    if (entry.sortOrder !== index) throw new Error('selection-manifest.json 的 sortOrder 必须从 0 连续递增。');
  });
  return { files: ordered, orderSource: 'selection-manifest-sortOrder' };
}

}
